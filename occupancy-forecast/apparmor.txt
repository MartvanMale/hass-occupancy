# AppArmor for the Occupancy Forecast add-on: reads the world, writes only to
# /data, /tmp and /run. The profile NAME below is a placeholder Supervisor
# rewrites to the installed slug; exactly ONE `^profile ` line, or
# get_profile_name() raises. A profile reloads on install and update, NEVER on
# rebuild -- which is why deploy-edge.sh routes through `ha addons update`.
# THIS PROFILE ENFORCES: a wrong rule stops the add-on on every installation.
# If enforcing breaks startup, add `complain` back -- do NOT uninstall: that removes /data.

#include <tunables/global>

profile occupancy_forecast flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Supervisor, the broker and optionally InfluxDB. abstractions/base does not grant this.
  network,

  # Docker grants these; AppArmor must agree. setuid/setgid for the s6-setuidgid drop
  # in run.sh (s6 is PID 1 and starts as root), the rest for its `chown -R /data`.
  capability setuid,
  capability setgid,
  capability chown,
  capability fowner,
  capability fsetid,
  capability dac_override,
  capability kill,

  # s6 supervises; it signals its children.
  signal (send,receive),

  # Read anything: the confinement here is about writes and execution.
  /** r,

  # Execute anything in the image, inheriting this profile. NOT Ux or Px -- nothing
  # escapes to unconfined. Enumerating the interpreters cannot be made safe: s6
  # materialises paths at runtime and its versions move with the base image.
  /** ix,

  # Write only where this add-on owns state: /data is the archive and the
  # models, /tmp is HOME and MPLCONFIGDIR, /run is s6's scratch.
  # `l` IS LOAD-BEARING: sem_open() hard-links, so without it joblib silently
  # falls back to threading. /data ITSELF as well as /data/**, for run.sh's chown.
  /data rwkl,
  /data/ rwkl,
  /data/** rwkl,
  /tmp/** rwkl,
  /run/** rwkl,
  /var/run/** rwkl,
  /dev/null rw,
  /dev/zero rw,
  /dev/full rw,
  /dev/random rw,
  /dev/urandom rw,
  /dev/tty rw,
  /dev/shm/** rwkl,

  # And nowhere else: the image is the deployment artifact. PYTHONDONTWRITEBYTECODE=1
  # keeps CPython off the /app and /opt rules.
  deny /etc/** w,
  deny /usr/** w,
  deny /bin/** w,
  deny /sbin/** w,
  deny /lib/** w,
  deny /opt/** w,
  deny /app/** w,
  deny /root/** rw,
}
